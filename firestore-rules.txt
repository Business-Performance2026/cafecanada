rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ===== Helper functions =====
    // isOwnerOrAdmin: EXACT same check used everywhere before — untouched.
    // hasEmployeePermission: NEW — true only if the caller is a registered
    // employee of this shop AND the owner has granted them this specific tab.
    // canManage: owner/admin always passes (unchanged behavior); employees
    // pass only if granted that specific permission.
    function isOwnerOrAdmin(shopId) {
      return request.auth != null && (
        request.auth.uid == shopId ||
        exists(/databases/$(database)/documents/admins/$(request.auth.uid))
      );
    }

    function hasEmployeePermission(shopId, key) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/shops/$(shopId)/employees/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/shops/$(shopId)/employees/$(request.auth.uid)).data.permissions[key] == true;
    }

    function canManage(shopId, key) {
      return isOwnerOrAdmin(shopId) || hasEmployeePermission(shopId, key);
    }

    // Admins collection: a user can only read their OWN admin record
    // (used just to check "is this person an admin?")
    match /admins/{adminId} {
      allow read: if request.auth != null && request.auth.uid == adminId;
      allow write: if false; // never editable from the client — add admins manually in Firebase Console
    }

    // Lets a logged-in employee find out which shop they belong to (their UID
    // alone doesn't reveal that). Only they can read their own entry; only the
    // owning shop or an admin can create/delete it (done when adding/removing staff).
    match /employeeIndex/{empUid} {
      allow read: if request.auth != null && request.auth.uid == empUid;
      allow create: if request.auth != null && request.resource.data.shopId != null &&
        isOwnerOrAdmin(request.resource.data.shopId);
      allow delete: if request.auth != null && isOwnerOrAdmin(resource.data.shopId);
      allow update: if false;
    }

    match /shops/{shopId} {
      allow read: if true;

      // Write allowed if: you own this shop, you're a registered admin, OR
      // you're an employee this owner granted the "account" permission to.
      // Exception: anyone (including a customer with no login) may update
      // ONLY the "info.lastOrderAt" field, and only to a timestamp — used
      // to track whether a shop is actively receiving orders, for the
      // admin panel. No other field can be touched through this exception.
      allow write: if canManage(shopId, 'account') || (
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['info.lastOrderAt']) &&
        request.resource.data.info.lastOrderAt is timestamp
      );

      match /menu/{itemId} {
        allow read: if true;
        // Either the "لوحة التحكم" (dashboard) or "المخزون" (inventory) tab
        // permission grants menu-item write access, since both tabs edit
        // this same collection (item details vs. just the stock count).
        allow write: if canManage(shopId, 'dashboard') || canManage(shopId, 'inventory');
      }

      match /orders/{orderId} {
        // Anyone can create an order (the customer, no login needed)
        allow create: if true;

        // Anyone who already knows the specific order ID can read just that
        // one order — this is what powers the customer's live status tracker,
        // which works without requiring the customer to log in.
        allow get: if true;

        // Owner/admin, or an employee with "نقطة البيع" or "المبيعات" access,
        // can list/query all orders at once.
        allow list: if canManage(shopId, 'pos') || canManage(shopId, 'sales');

        // Only the shop owner or an admin can change an order's status or delete it.
        // Exception: the customer (no login) may update ONLY the "rating" field
        // of an order they already know the ID of — used for the post-pickup
        // star rating. They cannot touch any other field this way.
        allow update: if canManage(shopId, 'pos') || (
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['rating']) &&
          request.resource.data.rating is number &&
          request.resource.data.rating >= 1 &&
          request.resource.data.rating <= 5
        );

        allow delete: if isOwnerOrAdmin(shopId);
      }

      // Admin-only private notes about this shop — NOT covered by the shop's
      // public read:true rule, since this lives in its own subcollection.
      // Employees never see this, regardless of permissions.
      match /adminNotes/{noteId} {
        allow read, write: if request.auth != null &&
          exists(/databases/$(database)/documents/admins/$(request.auth.uid));
      }

      // Admin-only payment records for this shop. Employees never see this.
      match /payments/{paymentId} {
        allow read, write: if request.auth != null &&
          exists(/databases/$(database)/documents/admins/$(request.auth.uid));
      }

      // Bookable rooms/halls — public read (customers browse them); managing
      // them is part of the "لوحة التحكم" (dashboard) tab.
      match /rooms/{roomId} {
        allow read: if true;
        allow write: if canManage(shopId, 'dashboard');
      }

      // Full reservation details (customer name, phone, notes) — private, same
      // pattern as orders: anyone can create one, dashboard-permission manages them.
      match /reservations/{resId} {
        allow create: if true;
        allow get: if true;
        allow list, update, delete: if canManage(shopId, 'dashboard');
      }

      // Minimal, PII-free markers used only to check "is this room/date/time
      // already taken?" before confirming a booking. Publicly readable/creatable
      // since they contain no personal data — just room+date+time.
      match /bookedSlots/{slotId} {
        allow read: if true;
        allow create: if request.resource.data.keys().hasOnly(['roomId', 'date', 'time', 'createdAt']);
        allow update: if false;
        allow delete: if canManage(shopId, 'dashboard');
      }

      // Staff accounts. The owner/admin manages the list; an employee may
      // read only their OWN record (to load their permissions on login).
      match /employees/{empUid} {
        allow get: if request.auth != null && (request.auth.uid == empUid || isOwnerOrAdmin(shopId));
        allow list: if isOwnerOrAdmin(shopId);
        allow write: if isOwnerOrAdmin(shopId);
      }

      // Loyalty points, keyed by customer phone number. Public read so a
      // customer can see their own progress before ordering; only staff with
      // "pos" access can add to or reset the count.
      match /loyaltyCustomers/{phone} {
        allow read: if true;
        allow write: if canManage(shopId, 'pos');
      }

      // Gift cards. Create is HARDENED: must include a receipt, a positive
      // total, at least one item, and start as not-redeemed.
      // Redeem: staff with "pos" access, OR the recipient self-claims
      // (false -> true only, no other field can change).
      match /giftCards/{giftId} {
        allow read: if true;
        allow create: if request.resource.data.receiptUrl is string &&
          request.resource.data.receiptUrl.size() > 0 &&
          request.resource.data.total is number &&
          request.resource.data.total > 0 &&
          request.resource.data.items is list &&
          request.resource.data.items.size() > 0 &&
          request.resource.data.redeemed == false;
        allow update: if canManage(shopId, 'pos') || (
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['redeemed', 'redeemedAt']) &&
          resource.data.redeemed == false &&
          request.resource.data.redeemed == true
        );
        allow delete: if isOwnerOrAdmin(shopId);
      }
    }

    // Global category icon rules (e.g. "hot coffee" -> a specific image),
    // managed by the admin and used across ALL shops' customer menus.
    // Public read since customers need to see them; only admins can manage them.
    match /categoryIcons/{iconId} {
      allow read: if true;
      allow write: if request.auth != null &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
  }
}
